# ── PyTorch: CPU-only wheels ──────────────────────────────────────────────
# Render web services have no GPU. This extra index makes pip pull the
# `+cpu` build of torch, which has NONE of the ~3 GB nvidia-cuda-* / triton
# dependencies the default PyPI wheel drags in on Linux. `pip install -r
# requirements.txt` alone is enough — no special build command needed.
--extra-index-url https://download.pytorch.org/whl/cpu

# ── Web framework ─────────────────────────────────────────────────────────
fastapi==0.115.6
uvicorn[standard]==0.32.1
python-multipart==0.0.19          # FastAPI needs this for multipart/form-data (PDF upload)
pydantic==2.10.3
python-dotenv==1.0.1

# ── PDF parsing ──────────────────────────────────────────────────────────
PyMuPDF==1.26.3                    # cp39-abi3 wheels → works on Python 3.9–3.14

# ── Embeddings + vector search (resume chat / RAG) ───────────────────────
# torch 2.9.0 is the oldest release still published on the PyTorch CPU index
# (older ones, incl. 2.2.x, were removed — that is what broke the build).
# sentence-transformers 5.1.1 is bumped to match; it stays on transformers 4.x
# (torch>=1.11 requirement satisfied) so no model / API behaviour changes.
torch==2.9.0+cpu; sys_platform == "linux"
torch==2.9.0; sys_platform != "linux"   # macOS / Windows dev machines (already CPU-only)
sentence-transformers==5.1.1
faiss-cpu==1.12.0                 # numpy<3,>=1.25 · cp39–cp314 wheels
numpy==2.1.3                      # torch 2.9 is built against numpy 2.x

# ── LLM (Gemini) ─────────────────────────────────────────────────────────
# google-generativeai is EOL (no updates/bug fixes) and pins the legacy
# v1beta endpoint where gemini-1.5-* now 404s. google-genai is the current
# maintained SDK. 2.8.0 is the newest release that still allows pydantic 2.10.x
# (2.9.0+ requires pydantic>=2.12.5) — keeps the rest of the stack unchanged.
google-genai==2.8.0
